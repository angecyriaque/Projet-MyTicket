    // Generates a deterministic custom QR grid based on registration code hash
    function generateQRGrid(ticketId) {
        const svg = document.getElementById('ticket-qr');
        if (!svg) return;
        
        let hash = 0;
        for (let i = 0; i < ticketId.length; i++) {
            hash = ticketId.charCodeAt(i) + ((hash << 5) - hash);
        }
        
        let qrContent = '';
        // Draw top-left positioning square
        qrContent += `<rect x="5" y="5" width="22" height="22" fill="#0B132B" rx="2"/>`;
        qrContent += `<rect x="9" y="9" width="14" height="14" fill="white" rx="1"/>`;
        qrContent += `<rect x="12" y="12" width="8" height="8" fill="#0B132B" rx="0.5"/>`;
        
        // Draw top-right positioning square
        qrContent += `<rect x="73" y="5" width="22" height="22" fill="#0B132B" rx="2"/>`;
        qrContent += `<rect x="77" y="9" width="14" height="14" fill="white" rx="1"/>`;
        qrContent += `<rect x="80" y="12" width="8" height="8" fill="#0B132B" rx="0.5"/>`;
        
        // Draw bottom-left positioning square
        qrContent += `<rect x="5" y="73" width="22" height="22" fill="#0B132B" rx="2"/>`;
        qrContent += `<rect x="9" y="77" width="14" height="14" fill="white" rx="1"/>`;
        qrContent += `<rect x="12" y="80" width="8" height="8" fill="#0B132B" rx="0.5"/>`;
        
        // Populate random-looking grid bits dependent on seeded hash
        for (let row = 0; row < 13; row++) {
            for (let col = 0; col < 13; col++) {
                // Keep corner positioning squares pristine and unmodified
                if ((row < 4 && col < 4) || (row < 4 && col > 8) || (row > 8 && col < 4)) {
                    continue;
                }
                
                const factorValue = Math.abs(Math.sin(hash + (row * 13) + (col * 31)));
                if (factorValue > 0.45) {
                    const x = 5 + (col * 7);
                    const y = 5 + (row * 7);
                    qrContent += `<rect x="${x}" y="${y}" width="5" height="5" fill="#0B132B" rx="0.5"/>`;
                }
            }
        }
        svg.innerHTML = qrContent;
    }

        const nom = document.getElementById('form-nom').value.trim();
        const prenom = document.getElementById('form-prenom').value.trim();
        const contact1 = document.getElementById('form-contact1').value.trim();
        const contact2 = document.getElementById('form-contact2').value.trim();
        const email = document.getElementById('form-email').value.trim();
        const eglise = document.getElementById('form-eglise').value.trim();
        
        const customId = `MYT-${Math.floor(1000 + Math.random() * 9000)}`;
        
        const newTicket = {
            id: customId,
            nom: nom,
            prenom: prenom,
            contact1: contact1,
            contact2: contact2 || "",
            email: email,
            eglise: eglise,
            dateAdded: new Date().toISOString()
        };
        
        state.participants.unshift(newTicket);
        
        try {
            localStorage.setItem('myticket_participants_v3', JSON.stringify(state.participants));
        } catch(e) {
            console.warn("Storage quota or permissions blocked writing.", e);
        }
        
        state.currentTicket = newTicket;
        updateTicketPreviewUI(newTicket);
        
        // Toggle download/print buttons status
        const btnPrint = document.getElementById('btn-print-ticket');
        const btnDownload = document.getElementById('btn-download-ticket');
        btnPrint.removeAttribute('disabled');
        btnPrint.classList.remove('opacity-50', 'cursor-not-allowed');
        btnDownload.removeAttribute('disabled');
        btnDownload.classList.remove('opacity-50', 'cursor-not-allowed');
        
        document.getElementById('ticket-generation-form').reset();
        
        showNotification(
            "Inscription validée !", 
            `Le ticket de ${prenom} ${nom} est prêt. Réf: ${customId}`, 
            "success"
        );
    }

    function updateTicketPreviewUI(ticket) {
        document.getElementById('ticket-display-name').innerText = `${ticket.prenom} ${ticket.nom}`;
        document.getElementById('ticket-display-church').innerText = ticket.eglise;
        
        const contactsText = ticket.contact2 ? `${ticket.contact1} / ${ticket.contact2}` : ticket.contact1;
        document.getElementById('ticket-display-contact').innerText = contactsText;
        document.getElementById('ticket-display-email').innerText = ticket.email;
        document.getElementById('ticket-display-ref').innerText = `#${ticket.id}`;
        
        // Update QR code dynamically with seeded id
        generateQRGrid(ticket.id);
        
        const statusBadge = document.getElementById('ticket-display-status');
        statusBadge.className = "text-[9px] font-bold px-2.5 py-1 rounded-full bg-emerald-100 text-emerald-700 uppercase tracking-wide";
        statusBadge.innerText = "CONFIRMÉ";
    }

    function printTicket() {
        if (!state.currentTicket) return;
        try {
            window.print();
        } catch (e) {
            console.warn("Direct window printing is restricted in this sandboxed environment:", e);
            openPrintModal();
        }
    }

    function openPrintModal() {
        const modal = document.getElementById('print-helper-modal');
        const preview = document.getElementById('print-ticket-raw-preview');
        
        if (!state.currentTicket) return;
        
        const rawText = getFormattedTicketText();
        preview.innerText = rawText;
        modal.classList.remove('hidden');
    }

    function closePrintModal() {
        document.getElementById('print-helper-modal').classList.add('hidden');
    }

    function getFormattedTicketText() {
        return `==============================================
MYTICKET - FESTIVAL JEUNESSE CHRÉTIENNE 2026
==============================================
RÉFÉRENCE : #${state.currentTicket.id}
PARTICIPANT : ${state.currentTicket.prenom} ${state.currentTicket.nom}
ÉGLISE D'ORIGINE : ${state.currentTicket.eglise}
CONTACTS : ${state.currentTicket.contact1} ${state.currentTicket.contact2 ? '/ ' + state.currentTicket.contact2 : ''}
E-MAIL : ${state.currentTicket.email}
DATE DU FESTIVAL : Samedi 14 Novembre 2026 à 14h00
STATUT : ACCÈS CONFIRMÉ ENREGISTRÉ
==============================================`;
    }

    function copyTicketDetailsFallback() {
        const rawText = getFormattedTicketText();
        
        try {
            const textarea = document.createElement('textarea');
            textarea.value = rawText;
            textarea.style.position = 'fixed';
            document.body.appendChild(textarea);
            textarea.select();
            document.execCommand('copy');
            document.body.removeChild(textarea);
            
            showNotification(
                "Détails copiés !",
                "Les détails du pass d'accès ont été copiés au presse-papier !",
                "success"
            );
            closePrintModal();
        } catch (err) {
            console.error("Clipboard copy failed:", err);
            showNotification("Erreur", "La copie a échoué.", "error");
        }
    }

    function downloadTicketImage() {
        if (!state.currentTicket) return;
        const ticketInfo = getFormattedTicketText();
        
        try {
            const blob = new Blob([ticketInfo], { type: 'text/plain;charset=utf-8;' });
            const link = document.createElement("a");
            const url = URL.createObjectURL(blob);
            link.setAttribute("href", url);
            link.setAttribute("download", `Ticket_${state.currentTicket.prenom}_${state.currentTicket.nom}_${state.currentTicket.id}.txt`);
            link.style.visibility = 'hidden';
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            
            showNotification(
                "Exportation réussie",
                "Le pass d'accès numérique a été exporté !",
                "info"
            );
        } catch (err) {
            console.warn("Download blocks detected, executing auto fallback copy:", err);
            copyTicketDetailsFallback();
        }
    }

    function updateDashboardStats() {
        const total = state.participants.length;
        document.getElementById('stat-total-tickets').innerText = total;
        
        const churches = [...new Set(state.participants.map(p => p.eglise.trim().toLowerCase()))];
        document.getElementById('stat-total-churches').innerText = churches.length;
        
        const objective = 200;
        const rate = Math.min(Math.round((total / objective) * 100), 100);
        document.getElementById('stat-rate').innerText = `${rate}%`;
        document.getElementById('stat-progress-bar').style.width = `${rate}%`;
        
        const counts = {};
        state.participants.forEach(p => {
            const ch = p.eglise.trim();
            counts[ch] = (counts[ch] || 0) + 1;
        });
        
        const sortedChurches = Object.entries(counts)
            .sort((a,b) => b[1] - a[1])
            .slice(0, 4);
            
        const topContainer = document.getElementById('dashboard-top-churches');
        topContainer.innerHTML = '';
        
        if (sortedChurches.length === 0) {
            topContainer.innerHTML = `<p class="text-xs text-slate-400 italic">Aucune donnée disponible</p>`;
        } else {
            sortedChurches.forEach(([name, count]) => {
                const percentage = total > 0 ? Math.round((count / total) * 100) : 0;
                topContainer.innerHTML += `
                    <div class="space-y-1">
                        <div class="flex items-center justify-between text-sm">
                            <div class="flex items-center gap-2">
                                <span class="w-2.5 h-2.5 rounded-full bg-brand-orange"></span>
                                <span class="font-semibold text-slate-700 truncate max-w-[140px] sm:max-w-none">${name}</span>
                            </div>
                            <span class="text-xs font-bold text-slate-500">${count} (${percentage}%)</span>
                        </div>
                        <div class="w-full bg-slate-100 h-1 rounded-full overflow-hidden">
                            <div class="bg-brand-orange h-full rounded-full" style="width: ${percentage}%"></div>
                        </div>
                    </div>
                `;
            });
        }
        
        const currentHour = new Date().toLocaleTimeString('fr-FR', {hour: '2-digit', minute:'2-digit'});
        document.getElementById('dashboard-date').innerText = `Mise à jour : ${currentHour}`;
    }

    function uploadAvatarImage(event) {
        const file = event.target.files[0];
        if (file) {
            if (!file.type.startsWith('image/')) {
                showNotification("Format invalide", "Veuillez choisir un fichier image.", "error");
                return;
            }
            
            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');
                    
                    canvas.width = 128;
                    canvas.height = 128;
                    
                    const minEdge = Math.min(img.width, img.height);
                    const sourceX = (img.width - minEdge) / 2;
                    const sourceY = (img.height - minEdge) / 2;
                    
                    ctx.drawImage(img, sourceX, sourceY, minEdge, minEdge, 0, 0, 128, 128);
                    
                    const compressedDataUrl = canvas.toDataURL('image/jpeg', 0.7);
                    state.user.avatar = compressedDataUrl;
                    
                    try {
                        localStorage.setItem('myticket_user', JSON.stringify(state.user));
                        applyUserStatesUI();
                        showNotification(
                            "Photo mise à jour",
                            "Votre avatar a été enregistré avec succès !",
                            "success"
                        );
                    } catch(err) {
                        showNotification("Erreur", "Espace de stockage saturé sur votre navigateur.", "error");
                    }
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
        }
    }

    function renderParticipantsTable() {
        const query = document.getElementById('search-participants-input').value.toLowerCase().trim();
        const churchFilter = document.getElementById('filter-church-select').value;
        const tbody = document.getElementById('participants-table-body');
        
        const filtered = state.participants.filter(p => {
            const matchesQuery = !query || 
                p.nom.toLowerCase().includes(query) || 
                p.prenom.toLowerCase().includes(query) || 
                p.email.toLowerCase().includes(query) || 
                p.contact1.includes(query) || 
                (p.contact2 && p.contact2.includes(query)) ||
                p.eglise.toLowerCase().includes(query);
                
            const matchesChurch = !churchFilter || p.eglise === churchFilter;
            
            return matchesQuery && matchesChurch;
        });
        
        tbody.innerHTML = '';
        
        if (filtered.length === 0) {
            document.getElementById('table-empty-state').classList.remove('hidden');
            document.getElementById('table-showing-text').innerText = "Affichage de 0 participants";
            return;
        }
        
        document.getElementById('table-empty-state').classList.add('hidden');
        document.getElementById('table-showing-text').innerText = `Affichage de ${filtered.length} participant(s) sur ${state.participants.length}`;
        
        filtered.forEach(p => {
            const contactsList = p.contact2 ? `${p.contact1} / ${p.contact2}` : p.contact1;
            tbody.innerHTML += `
                <tr class="hover:bg-slate-50/50 transition border-b border-slate-50">
                    <td class="py-4 px-6">
                        <span class="font-mono font-bold text-xs text-brand-dark bg-slate-100 px-2 py-1 rounded-md">${p.id}</span>
                    </td>
                    <td class="py-4 px-6 font-semibold text-slate-800">
                        <div class="flex flex-col">
                            <span>${p.prenom} ${p.nom}</span>
                            <span class="text-[10px] text-slate-400 font-normal">Inscrit le : ${new Date(p.dateAdded).toLocaleDateString('fr-FR')}</span>
                        </div>
                    </td>
                    <td class="py-4 px-6 text-slate-600 font-medium">${p.eglise}</td>
                    <td class="py-4 px-6 text-slate-600 font-mono text-xs">${contactsList}</td>
                    <td class="py-4 px-6 text-slate-500">${p.email}</td>
                    <td class="py-4 px-6 text-right space-x-1.5 whitespace-nowrap">
                        <button onclick="viewAttendeeTicket('${p.id}')" class="p-1.5 text-slate-500 hover:text-brand-orange bg-slate-100 hover:bg-orange-50 rounded-lg transition" title="Voir le ticket">
                            <i data-lucide="eye" class="h-4.5 w-4.5"></i>
                        </button>
                        <button onclick="deleteParticipant('${p.id}')" class="p-1.5 text-slate-400 hover:text-red-600 hover:bg-red-50 rounded-lg transition" title="Supprimer">
                            <i data-lucide="trash-2" class="h-4.5 w-4.5"></i>
                        </button>
                    </td>
                </tr>
            `;
        });
        
        initializeLucideIcons();
    }

    function populateChurchFilterOptions() {
        const select = document.getElementById('filter-church-select');
        const currentValue = select.value;
        const uniqueChurches = [...new Set(state.participants.map(p => p.eglise.trim()))].sort();
        
        select.innerHTML = '<option value="">Toutes les Églises</option>';
        uniqueChurches.forEach(church => {
            select.innerHTML += `<option value="${church}" ${currentValue === church ? 'selected' : ''}>${church}</option>`;
        });
    }

    function filterParticipants() {
        renderParticipantsTable();
    }

    function deleteParticipant(participantId) {
        state.participants = state.participants.filter(p => p.id !== participantId);
        try {
            localStorage.setItem('myticket_participants_v3', JSON.stringify(state.participants));
        } catch(e) {
            console.warn("Storage syncing blocked after record deletion.");
        }
        
        renderParticipantsTable();
        populateChurchFilterOptions();
        
        showNotification(
            "Suppression effectuée",
            `Le pass de référence ${participantId} a été supprimé.`,
            "info"
        );
    }

    function viewAttendeeTicket(participantId) {
        const ticket = state.participants.find(p => p.id === participantId);
        if (ticket) {
            state.currentTicket = ticket;
            updateTicketPreviewUI(ticket);
            switchView('ticket');
            
            const btnPrint = document.getElementById('btn-print-ticket');
            const btnDownload = document.getElementById('btn-download-ticket');
            btnPrint.removeAttribute('disabled');
            btnPrint.classList.remove('opacity-50', 'cursor-not-allowed');
            btnDownload.removeAttribute('disabled');
            btnDownload.classList.remove('opacity-50', 'cursor-not-allowed');
            
            showNotification(
                "Ticket Chargé",
                `Ticket #${ticket.id} affiché pour modification/impression.`,
                "info"
            );
        }
    }

    function clearHistory() {
        if (state.participants.length > 0) {
            state.participants = [];
            try {
                localStorage.setItem('myticket_participants_v3', JSON.stringify([]));
            } catch(e) {
                console.warn("Clearing database from local storage is prohibited.");
            }
            renderHistoryList();
            showNotification(
                "Historique réinitialisé",
                "Tous les participants enregistrés ont été effacés.",
                "info"
            );
        }
    }

    function exportToCsv() {
        if (state.participants.length === 0) {
            showNotification("Erreur", "Aucune donnée disponible à exporter.", "error");
            return;
        }
        
        let csvContent = "data:text/csv;charset=utf-8,";
        csvContent += "ID;Nom;Prenom;Contact1;Contact2;Email;Eglise;DateInscription\r\n";
        
        state.participants.forEach(p => {
            const row = `${p.id};${p.nom};${p.prenom};${p.contact1};${p.contact2 || ''};${p.email};${p.eglise};${p.dateAdded}`;
            csvContent += row + "\r\n";
        });
        
        try {
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "Participants_Festival_Jeunesse_2026.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            
            showNotification(
                "Exportation réussie !",
                "Le fichier d'extraction Excel/CSV a été téléchargé.",
                "success"
            );
        } catch (err) {
            showNotification("Erreur d'exportation", "L'accès au téléchargement a été refusé par le navigateur.", "error");
        }
    }
